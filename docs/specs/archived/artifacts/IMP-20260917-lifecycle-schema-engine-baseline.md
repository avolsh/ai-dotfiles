## Recorder

```bash
#!/usr/bin/env bash
# record.sh VALIDATOR_DIR OUT.jsonl — run both corpora, validate-specs.test.sh and
# lifecycle-mutations.test.sh against the validate-specs.py in VALIDATOR_DIR, logging
# every invocation's findings (normalised: temp roots → project name) to OUT.jsonl.
set -euo pipefail
VDIR="$(cd "$1" && pwd)"; OUT="$(cd "$(dirname "$2")" && pwd)/$(basename "$2")"
DOT="$(git -C "$(dirname "$0")" rev-parse --show-toplevel 2>/dev/null || pwd)"
DOT="${AI_DOTFILES:-$DOT}"
WS="$(cd "$DOT/../.." && pwd)"
W="$(mktemp -d)"; trap 'rm -rf "$W"' EXIT
: > "$OUT"
cat > "$W/rec.py" <<PY
import json, os, re, subprocess, sys
norm = lambda x: re.sub(r"(/private)?/(var|tmp)/[^ ]*?/tmp\.[A-Za-z0-9]+/", "<tmp>/", x)
V = "$VDIR/validate-specs.py"
real = subprocess.run([sys.executable, V, *sys.argv[1:]], capture_output=True, text=True)
args, it = [], iter(sys.argv[1:])
for a in it:
    if a == "--report": next(it, None); continue
    if a != "--json": args.append(a)
j = subprocess.run([sys.executable, V, *args, "--json"], capture_output=True, text=True)
try:
    rep = json.loads(j.stdout); root = rep["root"]
    tag = os.path.basename(root) if root.startswith(("/tmp", "/var", "/private")) else root.replace("$WS/", "")
    rows = sorted(f"{f['path']}:{f['line']}:{f['check']}:{f['message']}".replace(root, "<root>") for f in rep["findings"])
    rows = [norm(r) for r in rows]
    rec = {"root": tag, "argv": [norm(a) for a in args], "rc": real.returncode, "findings": rows}
except Exception as exc:
    rec = {"argv": [norm(a) for a in args], "rc": real.returncode, "error": f"{type(exc).__name__}: {exc}", "stderr": j.stderr[-400:]}
with open("$OUT", "a") as fh:
    fh.write(json.dumps(rec, ensure_ascii=False) + "\n")
sys.stdout.write(real.stdout); sys.stderr.write(real.stderr); sys.exit(real.returncode)
PY
python3 "$W/rec.py" "$DOT" >/dev/null 2>&1 || true
python3 "$W/rec.py" "$WS/src/github.com/tobeverse/tobevisit-content" >/dev/null 2>&1 || true
for t in validate-specs lifecycle-mutations; do
  sed -e "s|^VALIDATOR=.*|VALIDATOR=\"$W/rec.py\"|" \
      -e "s|^SCRIPTS=.*|SCRIPTS=\"$DOT/scripts\"|" \
      -e "s|\$(cd \"\$(dirname \"\$0\")/../..\" \&\& pwd)|$DOT|" \
      "$DOT/scripts/test/$t.test.sh" > "$W/$t.test.sh"
  # RECORD_ALLOW_FAIL=1: an older validator fails newer self-tests; record its findings anyway.
  bash "$W/$t.test.sh" >/dev/null 2>"$W/$t.err" || [ -n "${RECORD_ALLOW_FAIL:-}" ] \
    || { echo "$t FAILED under recorder:" >&2; tail -5 "$W/$t.err" >&2; exit 2; }
done
python3 - "$OUT" <<'PY'
import json, sys
rs = [json.loads(l) for l in open(sys.argv[1])]
print(f"invocations={len(rs)} findings={sum(len(r.get('findings', [])) for r in rs)} errors={sum('error' in r for r in rs)}")
PY
```
